cmake_minimum_required(VERSION 3.22.1)
project(llmapp)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_ROOT ${CMAKE_CURRENT_LIST_DIR}/../../../../llama.cpp)
set(LLAMA_BUILD ${CMAKE_CURRENT_LIST_DIR}/../../../../llama-build)
include_directories(${LLAMA_ROOT}/include ${LLAMA_ROOT}/ggml/include)
add_library(llmapp SHARED llmapp_jni.cpp)
add_library(llama STATIC IMPORTED)
set_target_properties(llama PROPERTIES IMPORTED_LOCATION ${LLAMA_BUILD}/src/libllama.a INTERFACE_INCLUDE_DIRECTORIES ${LLAMA_ROOT}/include)
add_library(ggml STATIC IMPORTED)
set_target_properties(ggml PROPERTIES IMPORTED_LOCATION ${LLAMA_BUILD}/ggml/src/libggml.a INTERFACE_INCLUDE_DIRECTORIES ${LLAMA_ROOT}/ggml/include)
add_library(ggml-base STATIC IMPORTED)
set_target_properties(ggml-base PROPERTIES IMPORTED_LOCATION ${LLAMA_BUILD}/ggml/src/libggml-base.a)
add_library(ggml-cpu STATIC IMPORTED)
set_target_properties(ggml-cpu PROPERTIES IMPORTED_LOCATION ${LLAMA_BUILD}/ggml/src/libggml-cpu.a)
add_library(kleidiai STATIC IMPORTED)
set_target_properties(kleidiai PROPERTIES IMPORTED_LOCATION ${LLAMA_BUILD}/_deps/kleidiai-build/libkleidiai.a)
target_link_libraries(llmapp PRIVATE llama ggml ggml-base ggml-cpu kleidiai log android atomic dl m)
